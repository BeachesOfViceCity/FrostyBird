cmake_minimum_required(VERSION 3.22)
project(frosty_guest LANGUAGES CXX)
if(NOT MINGW OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Use cmake/mingw64.cmake to build the x86-64 guest modules")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED YES)
set(CMAKE_CXX_EXTENSIONS NO)
option(FROSTY_COMPLETE_GUEST "Require every normal-play provider and its build-time inputs" OFF)
add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>")
add_link_options(-static)

set(XS_AUDIO_FFMPEG "${CMAKE_CURRENT_SOURCE_DIR}/../build/deps/audio-win64" CACHE PATH "Shared PE FFmpeg with the SHAPE XMA2 frame adapter")
if(FROSTY_COMPLETE_GUEST)
    foreach(input lib/libavcodec.dll.a lib/libavutil.dll.a bin/avcodec-62.dll bin/avutil-60.dll include/libavcodec/avcodec.h)
        if(NOT EXISTS "${XS_AUDIO_FFMPEG}/${input}")
            message(FATAL_ERROR "Missing ${XS_AUDIO_FFMPEG}/${input}; build audio-codec first")
        endif()
    endforeach()
endif()
if(EXISTS "${XS_AUDIO_FFMPEG}/lib/libavcodec.dll.a")
    add_library(acphal SHARED acphal.cpp shape_xma.cpp acphal.def)
    set_target_properties(acphal PROPERTIES PREFIX "" OUTPUT_NAME "AcpHal")
    target_include_directories(acphal SYSTEM PRIVATE "${XS_AUDIO_FFMPEG}/include")
    target_link_libraries(acphal PRIVATE "${XS_AUDIO_FFMPEG}/lib/libavcodec.dll.a" "${XS_AUDIO_FFMPEG}/lib/libavutil.dll.a")
    foreach(lib avcodec-62 avutil-60)
        configure_file("${XS_AUDIO_FFMPEG}/bin/${lib}.dll" "${CMAKE_CURRENT_BINARY_DIR}/${lib}.dll" COPYONLY)
    endforeach()
    add_executable(xs-shape-check test_shape.cpp)
    target_link_libraries(xs-shape-check PRIVATE acphal)
else()
    message(STATUS "For Xbox audio / Play, run tools/build_audio_codec.py with FFmpeg 8.1.2 sources and reconfigure")
endif()

    target_sources(d3d12_xs PRIVATE pass_times.cpp)
    target_compile_definitions(d3d12_xs PRIVATE FROSTY_PASS_PROFILE=1)

    CACHE FILEPATH "x64 PE zlib archive")

target_link_libraries(xcrdapi PRIVATE bcrypt)
set_target_properties(xcrdapi PROPERTIES PREFIX "")
set(FROSTY_LIBDEFLATE_SOURCE "" CACHE PATH "Optional libdeflate source tree for the guest DirectStorage decoder")
if(FROSTY_LIBDEFLATE_SOURCE)
    enable_language(C)
    set(CMAKE_POLICY_DEFAULT_CMP0077 NEW)
    set(LIBDEFLATE_BUILD_STATIC_LIB ON)
    set(LIBDEFLATE_BUILD_SHARED_LIB OFF)
    set(LIBDEFLATE_BUILD_GZIP OFF)
    set(LIBDEFLATE_COMPRESSION_SUPPORT OFF)
    set(LIBDEFLATE_GZIP_SUPPORT OFF)
    set(LIBDEFLATE_BUILD_TESTS OFF)
    add_subdirectory("${FROSTY_LIBDEFLATE_SOURCE}" libdeflate EXCLUDE_FROM_ALL)
    target_link_libraries(xcrdapi PRIVATE libdeflate::libdeflate_static)
    target_compile_definitions(xcrdapi PRIVATE FROSTY_FAST_INFLATE=1)
elseif(FROSTY_COMPLETE_GUEST)
    message(FATAL_ERROR "The complete guest requires FROSTY_LIBDEFLATE_SOURCE")
else()
    target_include_directories(xcrdapi PRIVATE "${FROSTY_WINE_SOURCE}/libs/zlib")
    target_link_libraries(xcrdapi PRIVATE "${FROSTY_GUEST_ZLIB}")
endif()

    target_compile_definitions(frosty-ray-rows PRIVATE FROSTY_VK_RAY_ROWS=1)

    message(FATAL_ERROR "Vulkan ray rows need glslang and ${FROSTY_WINE_SOURCE}/include/wine/vulkan.h")
