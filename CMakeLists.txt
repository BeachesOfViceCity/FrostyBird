cmake_minimum_required(VERSION 3.22.1)
# Existing host-only build trees keep their original targets.
set(full_build ON)
if(CMAKE_PROJECT_NAME STREQUAL "frosty_runtime")
    set(full_build OFF)
endif()
option(FROSTY_SUPERBUILD "Build the Linux tools, Wine, graphics backends and x64 guest runtime together" ${full_build})
if(FROSTY_SUPERBUILD)
    project(frosty_superbuild LANGUAGES C CXX)
    include(cmake/RuntimeBuild.cmake)
    return()
endif()

project(frosty_runtime LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED YES)
set(CMAKE_CXX_EXTENSIONS NO)

find_package(ZLIB REQUIRED)
find_package(Vulkan REQUIRED)
find_package(lz4 QUIET CONFIG)
if(TARGET LZ4::lz4)
    set(lz4_target LZ4::lz4)
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(LZ4 REQUIRED IMPORTED_TARGET liblz4)
    set(lz4_target PkgConfig::LZ4)
endif()
find_package(LLVM REQUIRED CONFIG)
if(NOT TARGET LLVM OR NOT "AMDGPU" IN_LIST LLVM_TARGETS_TO_BUILD)
    message(FATAL_ERROR "Frosty needs shared LLVM with the AMDGPU disassembler. Use the runtime preset with -DFROSTY_BUILD_LLVM=ON to build it.")
endif()
set(DXC_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/build/deps/dxc" CACHE PATH "Linux DirectXShaderCompiler distribution")
find_path(DXC_INCLUDE dxc/dxcapi.h HINTS "${DXC_ROOT}/include" REQUIRED)
find_library(DXC_LIBRARY dxcompiler HINTS "${DXC_ROOT}/lib" REQUIRED)

add_library(frosty-translation STATIC src/archive.cpp src/shader.cpp src/isa.cpp src/recompile.cpp src/dxil.cpp src/rt_format.cpp src/rt_collection.cpp src/rt_lower.cpp src/rt_machine.cpp src/inline_trace.cpp)
add_executable(frosty-shader src/main.cpp src/audit.cpp)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS src/ray_fetch.hlsli src/ray_visibility.hlsli)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/ray_fetch.hlsli" XS_FETCH_HLSL)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/ray_visibility.hlsli" XS_VISIBILITY_HLSL)
configure_file(cmake/ray_fetch.hpp.in "${CMAKE_CURRENT_BINARY_DIR}/ray_fetch.hpp" @ONLY)
target_include_directories(frosty-translation PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" PUBLIC src)
target_compile_options(frosty-translation PRIVATE -Wall -Wextra -Wpedantic)
target_compile_options(frosty-shader PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(frosty-translation SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS} "${DXC_INCLUDE}")
target_link_libraries(frosty-translation PUBLIC ZLIB::ZLIB ${lz4_target} LLVM "${DXC_LIBRARY}")
target_link_libraries(frosty-shader PRIVATE frosty-translation Vulkan::Vulkan)
add_executable(frosty-material-index src/material_index.cpp src/archive.cpp src/shader.cpp)
target_compile_options(frosty-material-index PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(frosty-material-index PRIVATE ZLIB::ZLIB ${lz4_target})
option(FROSTY_BUILD_INSTALLER "Build the native Matrix setup and player" OFF)
if(FROSTY_BUILD_INSTALLER)
    add_subdirectory(installer)
endif()
